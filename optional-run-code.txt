Optional indexes: the profile feed is faster with two small indexes. I added them to schema.sql and freely.dbml; nothing was removed. You can run this safely more than once:

USE freely;
GO
IF NOT EXISTS (SELECT 1 FROM sys.indexes
               WHERE name = 'IX_posts_author_created' AND object_id = OBJECT_ID('dbo.posts'))
    CREATE INDEX IX_posts_author_created ON dbo.posts (author_user_id, created_at);

IF NOT EXISTS (SELECT 1 FROM sys.indexes
               WHERE name = 'IX_post_reposts_user_created' AND object_id = OBJECT_ID('dbo.post_reposts'))
    CREATE INDEX IX_post_reposts_user_created ON dbo.post_reposts (user_id, created_at);
GO

Database changes

- Two new tables: bookmark_collections and bookmark_collection_posts. A post can only be filed into a collection if it is already saved.

- Docs updated: I added them, plus one index on post_bookmarks (user_id, created_at), to schema.sql and freely.dbml. Nothing was removed.

- To apply: you need to run this once in SSMS. It can be run again safely:
USE freely;
GO

IF OBJECT_ID('dbo.bookmark_collections', 'U') IS NULL
CREATE TABLE dbo.bookmark_collections (
    id          bigint IDENTITY(1,1) NOT NULL,
    user_id     bigint        NOT NULL,
    name        nvarchar(100) NOT NULL,
    created_at  datetime2     NOT NULL CONSTRAINT DF_bookmark_collections_created DEFAULT (SYSUTCDATETIME()),
    CONSTRAINT PK_bookmark_collections PRIMARY KEY (id),
    CONSTRAINT FK_bookmark_collections_user FOREIGN KEY (user_id) REFERENCES dbo.users (id),
    CONSTRAINT UQ_bookmark_collections_user_name UNIQUE (user_id, name)
);
GO

IF OBJECT_ID('dbo.bookmark_collection_posts', 'U') IS NULL
CREATE TABLE dbo.bookmark_collection_posts (
    collection_id  bigint    NOT NULL,
    post_id        bigint    NOT NULL,
    user_id        bigint    NOT NULL,
    created_at     datetime2 NOT NULL CONSTRAINT DF_bookmark_collection_posts_created DEFAULT (SYSUTCDATETIME()),
    CONSTRAINT PK_bookmark_collection_posts PRIMARY KEY (collection_id, post_id),
    CONSTRAINT FK_bookmark_collection_posts_collection FOREIGN KEY (collection_id) REFERENCES dbo.bookmark_collections (id),
    CONSTRAINT FK_bookmark_collection_posts_bookmark FOREIGN KEY (post_id, user_id) REFERENCES dbo.post_bookmarks (post_id, user_id)
);
GO

IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'IX_bookmark_collection_posts_bookmark'
               AND object_id = OBJECT_ID('dbo.bookmark_collection_posts'))
    CREATE INDEX IX_bookmark_collection_posts_bookmark ON dbo.bookmark_collection_posts (post_id, user_id);

IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = 'IX_post_bookmarks_user_created'
               AND object_id = OBJECT_ID('dbo.post_bookmarks'))
    CREATE INDEX IX_post_bookmarks_user_created ON dbo.post_bookmarks (user_id, created_at);
GO
